# 🧱 1) BACKEND — DATABASE MODEL

## 📁 Create file:

`/server/models/LandingPageSettings.js`

```js
import mongoose from "mongoose";

const ButtonSchema = new mongoose.Schema({
  label: String,
  link: String,
  active: { type: Boolean, default: true }
});

const ProductSchema = new mongoose.Schema({
  title: String,
  description: String,
  image: String,
  link: String,
  active: { type: Boolean, default: true }
});

const LandingPageSchema = new mongoose.Schema({
  homeBar: {
    title: String,
    subtitle: String
  },

  homeBarButtons: [ButtonSchema],

  footer: [{
    label: String,
    link: String,
    active: Boolean
  }],

  platformProducts: [ProductSchema],

  supportCard: {
    title: String,
    message: String,
    buttonText: String,
    buttonLink: String,
    active: Boolean
  },

  topBackground: String,

  visibilityFlags: {
    showProducts: Boolean,
    showSupport: Boolean,
    showFooter: Boolean
  }

}, { timestamps: true });

export default mongoose.model("LandingPageSettings", LandingPageSchema);
```

# ⚙️ 2) BACKEND — ROUTES

## 📁 Create:

`/server/routes/landingPage.js`

```js
import express from "express";
import LandingPage from "../models/LandingPageSettings.js";

const router = express.Router();

// GET
router.get("/", async (req, res) => {
  const data = await LandingPage.findOne();
  res.json(data);
});

// UPDATE
router.put("/", async (req, res) => {
  const updated = await LandingPage.findOneAndUpdate(
    {},
    req.body,
    { new: true, upsert: true }
  );
  res.json(updated);
});

export default router;
```

  filename: (_, file, cb) => {

# 🎨 5) FRONTEND — ADMIN PAGE

## 📁 Create:

`/src/pages/admin/LandingPage.tsx`

```tsx
import { useEffect, useState } from "react";
import api from "@/lib/api";

export default function LandingPageAdmin(){

  const [data,setData] = useState<any>({});

  useEffect(()=>{
    api.get("/api/admin/landing-page")
      .then(res=>setData(res.data || {}));
  },[]);

  const save = async ()=>{
    await api.put("/api/admin/landing-page", data);
    alert("Saved");
  };

  return (
    <div>

      <h2>Landing Page Control</h2>

      <input
        value={data?.homeBar?.title || ""}
        onChange={e=>setData({
          ...data,
          homeBar: { ...data.homeBar, title: e.target.value }
        })}
      />

      <button onClick={save}>Save</button>

    </div>
  );
}
```
