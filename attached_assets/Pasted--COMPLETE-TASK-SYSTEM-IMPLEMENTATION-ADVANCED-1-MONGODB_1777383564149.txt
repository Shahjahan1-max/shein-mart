# 🧱 1) MONGODB SEED — 25 ADS PER PACKAGE

## 📁 File:

`/server/seed/taskPackages.js`

```js
const mongoose = require("mongoose");

const TaskSchema = new mongoose.Schema({
  packageId: String,
  title: String,
  description: String,
  price: Number,
  commission: Number,
  order: Number,
});

const Task = mongoose.model("Task", TaskSchema);

async function seedTasks() {
  const packages = ["amazon_201", "amazon_202", "amazon_301"];

  for (let pkg of packages) {
    let tasks = [];

    for (let i = 1; i <= 25; i++) {
      tasks.push({
        packageId: pkg,
        title: `Product Task ${i}`,
        description: `Complete task ${i} for ${pkg}`,
        price: Math.floor(Math.random() * 50) + 50,
        commission: 0.04, // 4%
        order: i,
      });
    }

    await Task.insertMany(tasks);
    console.log(`✅ Seeded 25 tasks for ${pkg}`);
  }

  process.exit();
}

seedTasks();
```

# 🧱 2) TASK PROGRESSION SYSTEM

---

## 📁 Model:

`/server/models/UserTaskProgress.js`

```js
const mongoose = require("mongoose");

const UserTaskProgressSchema = new mongoose.Schema({
  userId: mongoose.Schema.Types.ObjectId,
  packageId: String,
  currentIndex: { type: Number, default: 0 },
  completedTasks: [{ type: Number }],
  totalEarned: { type: Number, default: 0 },
  status: {
    type: String,
    enum: ["not_started", "in_progress", "completed"],
    default: "not_started",
  },
});

module.exports = mongoose.model("UserTaskProgress", UserTaskProgressSchema);
```

## 📁 API: COMPLETE TASK

`POST /api/tasks/complete`

```js
const { taskIndex } = req.body;

const progress = await UserTaskProgress.findOne({ userId, packageId });

if (!progress) return res.status(400).json({ error: "No progress found" });

// prevent duplicate
if (progress.completedTasks.includes(taskIndex)) {
  return res.json(progress);
}

const task = await Task.findOne({ packageId, order: taskIndex });

// calculate earning
const earning = task.price * task.commission;

progress.completedTasks.push(taskIndex);
progress.currentIndex = taskIndex;
progress.totalEarned += earning;

// check completion
if (progress.completedTasks.length === 25) {
  progress.status = "completed";
}

await progress.save();

res.json(progress);
```
