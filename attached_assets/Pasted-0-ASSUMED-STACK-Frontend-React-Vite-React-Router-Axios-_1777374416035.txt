2) BACKEND API
2.1 Create: /server/routes/extraTasks.js
import express from "express";
import ExtraTask from "../models/ExtraTask.js";

const router = express.Router();

// GET all
router.get("/", async (req,res)=>{
  const tasks = await ExtraTask.find();
  res.json(tasks);
});

// CREATE
router.post("/", async (req,res)=>{
  const task = await ExtraTask.create(req.body);
  res.json(task);
});

// ASSIGN
router.post("/assign", async (req,res)=>{
  const { taskId, userId } = req.body;
  const task = await ExtraTask.findByIdAndUpdate(
    taskId,
    { assignedTo: userId },
    { new: true }
  );
  res.json(task);
});

// DELETE
router.delete("/:id", async (req,res)=>{
  await ExtraTask.findByIdAndDelete(req.params.id);
  res.json({ ok:true });
});

export default router;

  const { taskId } = req.body;
  const task = await Task.findById(taskId);

  }

  task.order_status = "in_progress";
  await task.save();
